import { serve } from "https://deno.land/std@0.168.0/http/server.ts"; import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; serve(async (req) => { if (req.method !== "POST") { return new Response("Method not allowed", { status: 405 }); } const supabase = createClient( Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")! ); const token = req.headers.get("Authorization"); if (!token) { return Response.json({ error: "Chýba prihlásenie." }, { status: 401 }); } const userClient = createClient( Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_ANON_KEY")!, { global: { headers: { Authorization: token } } } ); const { data: authData, error: authError } = await userClient.auth.getUser(); if (authError || !authData.user) { return Response.json({ error: "Neplatné prihlásenie." }, { status: 401 }); } const { exhibition_id } = await req.json(); if (!exhibition_id) { return Response.json({ error: "Chýba výstava." }, { status: 400 }); } const userId = authData.user.id; const email = authData.user.email; const { data: profile } = await supabase .from("profiles") .select("first_name,last_name") .eq("id", userId) .single(); const { data: exhibition, error: exhibitionError } = await supabase .from("exhibitions") .select("name,start_date,location") .eq("id", exhibition_id) .single(); if (exhibitionError || !exhibition) { return Response.json({ error: "Výstava sa nenašla." }, { status: 404 }); } const { data: entries, error: entriesError } = await supabase .from("entries") .select("id,animal_id,animals(category,breed,animal_number)") .eq("exhibition_id", exhibition_id); if (entriesError) { return Response.json({ error: entriesError.message }, { status: 500 }); } const mojePrihlasky = (entries || []).filter((entry: any) => entry.animals && entry.animals.owner_id === userId ); if (!mojePrihlasky.length) { return Response.json({ error: "Na túto výstavu nemáte prihlásené žiadne zvieratá." }, { status: 400 }); } const { data: settings } = await supabase .from("app_settings") .select("entry_fee") .eq("id", 1) .single(); const cena = Number(settings?.entry_fee ?? 1); const spolu = mojePrihlasky.length * cena; const zoznam = mojePrihlasky.map((entry: any) => { const z = entry.animals; return `
  • ${z.category} – ${z.breed} – číslo: ${z.animal_number || "—"} – ${cena.toFixed(2)} €
    • `; }).join(""); const meno = [profile?.first_name, profile?.last_name].filter(Boolean).join(" "); const html = `

    Potvrdenie prihlášky na výstavu

    Dobrý deň ${meno || ""},

    Vaša prihláška na výstavu ${exhibition.name} bola prijatá.

    Zoznam prihlásených zvierat:

      ${zoznam}

    Počet zvierat: ${mojePrihlasky.length}

    Poplatok za kus: ${cena.toFixed(2)} €

    Celkový poplatok: ${spolu.toFixed(2)} €

    Ďakujeme za Vašu prihlášku.

    ZO SZCH Koniarovce

    `; const resendKey = Deno.env.get("RESEND_API_KEY"); if (!resendKey) { return Response.json({ error: "Nie je nastavený e-mailový kľúč RESEND_API_KEY." }, { status: 500 }); } const mail = await fetch("https://api.resend.com/emails", { method: "POST", headers: { "Authorization": `Bearer ${resendKey}`, "Content-Type": "application/json" }, body: JSON.stringify({ from: "ZO SZCH Koniarovce ", to: [email!], subject: `Potvrdenie prihlášky – ${exhibition.name}`, html }) }); const mailResult = await mail.json(); if (!mail.ok) { return Response.json({ error: mailResult.message || "E-mail sa nepodarilo odoslať." }, { status: 500 }); } return Response.json({ success: true, message: "Potvrdenie bolo odoslané na e-mail.", pocetZvierat: mojePrihlasky.length, celkovyPoplatok: spolu }); });
