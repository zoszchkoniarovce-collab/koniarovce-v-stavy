async function ukoncitPrihlasku(){ if(jeAdmin){ sprava('Administrátor nemôže odosielať prihlášky.'); return; } const exhibition_id=document.getElementById('vyberVystavy').value; if(!exhibition_id){ sprava('Najskôr vyberte výstavu.'); return; } sprava('Kontrolujem stav výstavy...'); const {data:vystava,error:kontrolaError}=await db .from('exhibitions') .select('id,name,status') .eq('id',exhibition_id) .maybeSingle(); if(kontrolaError){ sprava('Chyba pri kontrole výstavy: '+kontrolaError.message); return; } if(!vystava){ sprava('Výstava sa nenašla. Obnovte stránku a skúste znova.'); return; } const stav=String(vystava.status||'') .trim() .toLowerCase(); if(stav==='zrušená'||stav==='zrusena'){ sprava('Táto výstava je zrušená.'); return; } if(!confirm('Odoslať potvrdenie prihlášky na výstavu „'+vystava.name+'“ na váš e-mail?')){ return; } sprava('Odosielam potvrdenie, chvíľu počkajte...'); const {data,error}=await db.functions.invoke( 'odoslat-potvrdenie', {body:{exhibition_id:vystava.id}} ); if(error){ sprava('Chyba pri odosielaní: '+error.message); return; } if(data?.error){ sprava('Chyba pri odosielaní: '+data.error); return; } sprava( data?.message|| 'Potvrdenie prihlášky bolo odoslané na váš e-mail. Celkový poplatok: '+ Number(data?.celkovyPoplatok||0).toFixed(2)+' €.' ); }
